/* --- Pillar 1: render ONLY the visible window (~40 rows, constant DOM) --- */
function svRenderWindow(force) {
  const scroller = $('sv-wrap'); if (!scroller) return;
  const total = SV.view.length;
  $('sv-spacer').style.height = (total * SV.ROW_H) + 'px';

  const vh = scroller.clientHeight, st = scroller.scrollTop;
  let start = Math.max(0, Math.floor(st / SV.ROW_H) - SV.OVERSCAN);
  let end   = Math.min(total, Math.ceil((st + vh) / SV.ROW_H) + SV.OVERSCAN);
  if (!force && start === SV.start && end === SV.end) return;
  SV.start = start; SV.end = end;

  const parts = new Array(end - start);
  for (let k = start; k < end; k++) {
    const r = SV.data[SV.view[k]];
    parts[k - start] =
      '
' +
      `
${svCell(r.service)}
` +
      `
${svCell(r.name)}
` +
      `
${svCell(r.category)}
` +
      `
${svCell(r.type)}
` +
      `
${svCell(r.rate)}
` +
      `
${svCell(r.min)}
` +
      `
${svCell(r.max)}
` +
      `
${svCell(r.refill)}
` +
      `
${svCell(r.cancel)}
` +
      '
';
  }
  $('sv-window').style.transform = `translateY(${start * SV.ROW_H}px)`;
  $('sv-window').innerHTML = parts.join('');
}

function svUpdateHeaderIndicators() {
  document.querySelectorAll('.sv-hcell.sortable').forEach(h => {
    const active = state.smm.servicesSort.column === h.dataset.col;
    h.classList.toggle('sorted', active);
    h.querySelector('.sort-ind').textContent = active ? (state.smm.servicesSort.dir === 'ASC' ? '▲' : '▼') : '⇅';
  });
}

function renderServicesUI() {
  const container = $('api-results');
  if (!container.querySelector('.sv-toolbar')) {
    container.innerHTML = `
      

        
        
        ⚡ Syncing latest...
      

      

        

          
${SV_COLS.map(c =>
            `
${SV_LABELS[c]}${SV_SORTABLE.has(c) ? '⇅' : ''}
`
          ).join('')}

          

        

      
`;

    /* Pillar 4: debounced search — 1 recompute per 120ms, not per keystroke */
    $('services-search').addEventListener('input', (e) => {
      state.smm.servicesSearch = e.target.value;
      $('services-match-count').textContent = 'Filtering…';
      clearTimeout(SV.searchTimer);
      SV.searchTimer = setTimeout(svRecompute, 120);
    });

    document.querySelectorAll('.sv-hcell.sortable').forEach(h => {
      h.addEventListener('click', () => {
        const s = state.smm.servicesSort, col = h.dataset.col;
        if (s.column === col) {
          if (s.dir === 'ASC') s.dir = 'DESC';
          else { s.column = null; s.dir = 'ASC'; }
        } else { s.column = col; s.dir = 'ASC'; }
        svUpdateHeaderIndicators();
        svRecompute();
      });
    });

    /* Pillar 4: rAF-throttled, passive scroll → max 1 window render per frame */
    $('sv-wrap').addEventListener('scroll', () => {
      if (!SV.raf) SV.raf = requestAnimationFrame(() => { SV.raf = 0; svRenderWindow(false); });
    }, { passive: true });
  }
  svUpdateHeaderIndicators();
  svRecompute();
}

/* --- SWR fetch, patched: preserves scroll, defers localStorage stringify --- */
async function handleServicesFetch() {
  const cacheKey = `swr_services_${state.activeAccountId}`;
  const cached = localStorage.getItem(cacheKey);
  $('api-raw').style.display = 'none';

  if (cached) {
    try {
      svSetData(JSON.parse(cached));                       // STALE: instant paint
      $('response-status').innerHTML = 'Cached • revalidating…';
    } catch (e) { localStorage.removeItem(cacheKey); }
  } else {
    $('api-results').innerHTML = '
Fetching services (first time)…
';
  }

  state.smm.servicesSyncing = true; updateSyncBadge();
  try {
    const fresh = await smmApi('services', {});
    if (fresh && fresh.error) throw new Error(fresh.error);
    if (!Array.isArray(fresh)) throw new Error('Invalid response format');

    /* Pillar 5: defer the ~2MB JSON.stringify OFF the interaction path */
    setTimeout(() => { try { localStorage.setItem(cacheKey, JSON.stringify(fresh)); } catch (e) {} }, 0);

    const scroller = $('sv-wrap');
    const prevScroll = scroller ? scroller.scrollTop : 0;   // preserve scroll + filters + sort
    svSetData(fresh);
    if ($('sv-wrap')) $('sv-wrap').scrollTop = Math.min(prevScroll, Math.max(0, SV.view.length * SV.ROW_H - $('sv-wrap').clientHeight));

    $('response-status').innerHTML = '✓ Synchronized';
    toast('Services synchronized (fresh data loaded)', 'success');
  } catch (err) {
    if (!cached && !$('sv-wrap')) $('api-results').innerHTML = `
Error
${escapeHtml(err.message)}
`;
    $('response-status').innerHTML = '✕ Sync Failed';
    toast('Error syncing: ' + err.message, 'error');
  } finally {
    state.smm.servicesSyncing = false; updateSyncBadge();
  }
}

function renderApiResults(data) {
  state.smm.lastResponse = data;
  const container = $('api-results');
  let rows = Array.isArray(data) ? data : [];
  if (!Array.isArray(data) && typeof data === 'object' && data !== null) {
    const keys = Object.keys(data);
    if (keys.length && keys.every(k => typeof data[k] === 'object' && data[k] !== null)) {
      rows = keys.map(k => ({ _key: k, ...data[k] }));
    } else {
      rows = [data];
    }
  }

  if (!rows.length) {
    container.innerHTML = '
Empty response.
';
    $('api-raw').textContent = JSON.stringify(data, null, 2);
    return;
  }

  const cols = [...new Set(rows.flatMap(r => Object.keys(r)))];
  let html = '';
  cols.forEach(c => html += ``);
  html += '';
  rows.forEach(r => {
    html += '';
    cols.forEach(c => {
      const v = r[c];
      const display = v === null || v === undefined ? 'NULL' :
        (typeof v === 'object' ? escapeHtml(JSON.stringify(v)) : escapeHtml(String(v)));
      html += ``;
    });
    html += '';
  });
  html += '
${escapeHtml(c)}
${display}
';
  container.innerHTML = html;
  $('api-raw').textContent = JSON.stringify(data, null, 2);
}

async function executeSmmAction(btn) {
  const action = state.smm.action;
  if (action === 'services') {
    await handleServicesFetch();
    return;
  }
  const cfg = ACTION_CONFIG[action];
  if (!cfg) return;

  const realAction = cfg.realAction || action;
  const params = {};
  const fields = cfg.hasServiceType ? (SERVICE_TYPE_FIELDS[state.smm.serviceType] || []) : cfg.fields;

  for (const f of fields) {
    const el = $(`smm-field-${f.name}`);
    if (el) {
      const val = el.value.trim();
      if (f.required && !val) {
        toast(`"${f.label}" is required`, 'error');
        el.focus();
        return;
      }
      if (val !== '') params[f.name] = val;
    }
  }

  try {
    const origText = btn.textContent;
    btn.disabled = true;
    btn.innerHTML = ' Requesting...';
    $('response-status').textContent = 'Executing request...';
    $('api-raw').style.display = 'none';

    const data = await smmApi(realAction, params);
    renderApiResults(data);
    $('response-status').textContent = 'Success (200 OK)';
    toast('Request executed successfully', 'success');
    btn.disabled = false;
    btn.textContent = origText;
  } catch (err) {
    renderApiResults({ error: err.message });
    $('response-status').textContent = 'Failed';
    toast('API error: ' + err.message, 'error');
    btn.disabled = false;
    btn.textContent = 'Execute Request';
  }
}



function bindEvents() {
  $('account-select').addEventListener('change', (e) => {
    state.activeAccountId = e.target.value;
    saveAccounts();
    renderProfileBar();
    initAfterProfileChange();
    toast(`Switched to ${getActive().name}`, 'info');
  });
  $('btn-add-account').addEventListener('click', () => openAccountForm());
  $('btn-manage-accounts').addEventListener('click', openAccountManager);
  $('modal-close').addEventListener('click', closeModal);
  $('modal-root').addEventListener('click', (e) => { if (e.target.id === 'modal-root') closeModal(); });

  document.querySelectorAll('.view-tab').forEach(t => {
    t.addEventListener('click', () => {
      document.querySelectorAll('.view-tab').forEach(x => x.classList.remove('active'));
      t.classList.add('active');
      state.activeView = t.dataset.view;
      document.querySelectorAll('.view-panel').forEach(p => p.classList.remove('active'));
      $(`view-${t.dataset.view}`).classList.add('active');
      $('sidebar').style.display = t.dataset.view === 'db' ? 'flex' : 'none';
      if (t.dataset.view === 'api' && state.smm.action === 'services' && !state.smm.servicesData.length) {
        handleServicesFetch();
      }
    });
  });

  $('btn-refresh-tables').addEventListener('click', loadTables);
  $('btn-refresh-data').addEventListener('click', loadData);
  $('btn-insert-row').addEventListener('click', openInsertModal);

  $('btn-apply-filter').addEventListener('click', () => {
    state.db.filter = {
      column: $('filter-column').value,
      op: $('filter-op').value,
      value: $('filter-value').value
    };
    state.db.page = 1;
    loadData();
  });
  $('btn-clear-filter').addEventListener('click', () => {
    state.db.filter = { column: '', op: 'LIKE', value: '' };
    $('filter-column').value = '';
    $('filter-value').value = '';
    state.db.page = 1;
    loadData();
  });
  $('filter-value').addEventListener('keydown', (e) => {
    if (e.key === 'Enter') $('btn-apply-filter').click();
  });

  $('btn-prev').addEventListener('click', () => {
    if (state.db.currentPage > 1) {
      state.db.currentPage--;
      runTablePipeline();
    }
  });
  $('btn-next').addEventListener('click', () => {
    const totalPages = Math.ceil(state.db.rawTableData.length / state.db.pageSize);
    if (state.db.currentPage < totalPages) {
      state.db.currentPage++;
      runTablePipeline();
    }
  });
  $('page-size').addEventListener('change', (e) => {
    state.db.pageSize = Number(e.target.value);
    state.db.currentPage = 1;
    runTablePipeline();
  });

  // SMM API events
  $('smm-action').addEventListener('change', (e) => {
    state.smm.action = e.target.value;
    renderSmmForm();
    if (e.target.value === 'services') {
      handleServicesFetch();
    }
  });

  $('btn-execute-smm').addEventListener('click', (e) => executeSmmAction(e.target));

  $('btn-toggle-raw').addEventListener('click', () => {
    const raw = $('api-raw');
    raw.style.display = raw.style.display === 'none' ? 'block' : 'none';
  });
}
